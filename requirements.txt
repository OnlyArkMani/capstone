# --- core ---
# PINNED for the same reason scikit-learn is, below. The fitted trust model
# records the sklearn, numpy and joblib versions that produced it and warns on
# load if any of them has moved, because a joblib pickle of sklearn estimators
# carries no cross-version guarantee. All three drifted on one machine
# (sklearn 1.9.0 -> 1.7.1, numpy 2.4.6 -> 1.26.2, joblib 1.6.0 -> 1.5.2) and
# every run printed "the trust scores from this run may be invalid" while
# producing plausible-looking numbers. Re-aligning the environment to these
# versions cleared the warning and left every detection-bearing figure
# unchanged, which is the evidence that the drift had been harmless -- but it
# was not knowable without checking, which is the point of the pins.
#
# Changing any of these three REQUIRES refitting: python -m fusion.train
numpy==2.4.6

# --- embeddings and retrieval (pipeline/) ---
# For a CPU-only install, install torch first from the CPU index:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
sentence-transformers>=3.0
faiss-cpu>=1.8

# --- detectors/ ---
# protectai/deberta-v3-base-prompt-injection-v2
transformers>=4.40
# cross-encoder/nli-deberta-v3-base
# (sentence-transformers, above, provides the CrossEncoder)
# Both models are DeBERTa-v3; their tokenizer needs sentencepiece.
sentencepiece>=0.2

# --- fusion/ : clustering, logistic regression, calibration, grouped CV ---
# PINNED, not a floor. The fitted trust model is a pickle containing sklearn
# estimator objects, and sklearn does not guarantee those load correctly across
# versions -- it warns "may lead to invalid results" and keeps going, so the
# failure is silent and the trust scores are simply wrong. A `>=` floor let a
# rebuild install a newer sklearn than the one that fitted fusion/artifacts/
# trust_model.joblib, and that is exactly what happened.
#
# Changing this version REQUIRES refitting: run `python -m fusion.train` (or the
# whole `verify` service) afterwards, or the mismatch returns.
scikit-learn==1.9.0
# Model persistence. Without joblib the fitted model in fusion/artifacts/ cannot be
# read: the composite trust score is absent from every report and only the rule
# track contributes to a disposition. The system degrades rather than failing, which
# is exactly why this omission is easy to miss.
joblib==1.6.0

streamlit>=1.35
